const search = document.getElementById('search'); const news = document.getElementById('news'); async function loadNews() { try { const response = await fetch( 'https://api.github.com/repos/sporttyqqazaq/sporttyqqazaq/contents/content/news' ); if (!response.ok) throw new Error('Жаңалықтар жүктелмеді'); const files = await response.json(); const posts = await Promise.all( files .filter(file => file.name.endsWith('.json')) .map(async file => { const result = await fetch(file.download_url); return result.json(); }) ); news.replaceChildren(); posts.sort((a, b) => (b.date || '').localeCompare(a.date || '')); posts.forEach(post => { const card = document.createElement('article'); card.className = 'card'; card.dataset.text = `${post.title} ${post.body}`; const visual = document.createElement('div'); visual.className = 'visual'; if (post.image) { const img = document.createElement('img'); img.src = post.image; img.alt = post.title || 'Жаңалық суреті'; img.style.cssText = 'width:100%;height:100%;object-fit:cover'; visual.appendChild(img); } else { visual.textContent = '🇰🇿'; } const body = document.createElement('div'); body.className = 'body'; const category = document.createElement('div'); category.className = 'category'; category.textContent = post.date || 'Жаңалық'; const title = document.createElement('h3'); title.textContent = post.title || ''; const text = document.createElement('p'); text.textContent = post.body || ''; body.append(category, title, text); card.append(visual, body); news.appendChild(card); }); } catch (error) { console.error(error); news.textContent = 'Жаңалықтарды жүктеу мүмкін болмады.'; } } search.addEventListener('input', () => { const q = search.value.toLocaleLowerCase('kk'); document.querySelectorAll('.card').forEach(card => { card.style.display = card.dataset.text.toLocaleLowerCase('kk').includes(q) ? '' : 'none'; }); }); loadNews();
